const DAILY_GOAL = 8 * 60 * 60;

  "Türkçe",

/* =========================
   TARİH
========================= */

function todayKey(){

  const d = new Date();

  const y = d.getFullYear();

  const m = String(d.getMonth()+1).padStart(2,"0");

  const day = String(d.getDate()).padStart(2,"0");

  return `${y}-${m}-${day}`;
}


/* =========================
   LOCAL STORAGE
========================= */

function getDay(key = todayKey()){

  const raw = localStorage.getItem("kpss_" + key);

  if(raw){

    try{
      return JSON.parse(raw);
    }catch(e){}
  }

  return {
    time:{},
    questions:{}
  };
}


function saveDay(data,key=todayKey()){

  localStorage.setItem(
    "kpss_" + key,
    JSON.stringify(data)
  );
}


/* =========================
   FORMAT
========================= */

function formatTime(seconds){

  seconds = Math.max(0,Math.round(seconds));

  const h = Math.floor(seconds/3600);

  const m = Math.floor((seconds%3600)/60);

  const s = seconds%60;

  if(h > 0){

    return `${String(h).padStart(2,"0")}:${String(m).padStart(2,"0")}:${String(s).padStart(2,"0")}`;

  }

  return `${String(m).padStart(2,"0")}:${String(s).padStart(2,"0")}`;
}


function formatMinutes(seconds){

  const mins = Math.round(seconds/60);

  if(mins < 60){

    return mins + " dk";
  }

  const h = Math.floor(mins/60);

  const m = mins%60;

  return `${h}s ${m}dk`;
}


/* =========================
   SERVICE WORKER
========================= */

async function registerSW(){

  if(!("serviceWorker" in navigator)){

    console.log("Service Worker desteklenmiyor.");

    return;
  }

  try{

    swReg = await navigator.serviceWorker.register(
      "./sw.js",
      {scope:"./"}
    );

    await navigator.serviceWorker.ready;

    console.log("Service Worker aktif.");

  }catch(error){

    console.log(
      "Service Worker kaydı başarısız:",
      error
    );

  }
}


/* =========================
   BİLDİRİM
========================= */

function notificationSupported(){

  return "Notification" in window;
}


function updateNotificationStatus(){

  const el =
    document.getElementById("notifyStatus");

  if(!notificationSupported()){

    el.textContent =
      "Bu tarayıcı bildirim desteklemiyor.";

    return;
  }

  if(Notification.permission === "granted"){

    el.textContent =
      "Bildirimler aktif ✓";

  }else if(Notification.permission === "denied"){

    el.textContent =
      "Bildirimler engellenmiş.";

  }else{

    el.textContent =
      "Bildirimler kapalı.";
  }
}


async function requestNotifications(){

  if(!notificationSupported()){

    alert(
      "Bu tarayıcı bildirimleri desteklemiyor."
    );

    return;
  }

  try{

    const permission =
      await Notification.requestPermission();

    updateNotificationStatus();

    if(permission === "granted"){

      await sendNotification(
        "KPSS Odak Merkezi 🎯",
        "Bildirimler aktif edildi."
      );

    }

  }catch(error){

    console.log(error);

    alert(
      "Bildirim izni alınamadı."
    );
  }
}


async function sendNotification(title,body){

  if(
    !notificationSupported() ||
    Notification.permission !== "granted"
  ){

    return;
  }

  try{

    const reg =
      swReg ||
      await navigator.serviceWorker.ready;

    if(reg && reg.active){

      reg.active.postMessage({

        type:"NOTIFY",

        title:title,

        body:body

      });

      return;
    }

  }catch(error){

    console.log(
      "SW bildirim hatası:",
      error
    );

  }


  /* Fallback */

  try{

    new Notification(
      title,
      {
        body:body,
        tag:"kpss-timer"
      }
    );

  }catch(error){

    console.log(error);
  }
}

/* =========================

function addQuestions(index){

  const input =
    document.getElementById(
      "q_" + index
    );

  const value =
    Number(input.value);

  if(!value || value < 1){

    alert(
      "Lütfen soru sayısını gir."
    );

    return;
  }

  const data = getDay();

  const subject = subjects[index];

  data.questions[subject] =
    (data.questions[subject] || 0) +
    value;

  saveDay(data);

  input.value = "";

  updateAll();

}


/* =========================
   TIMER BAŞLAT
========================= */

function startTimer(index,total){

  if(timer){

    alert(
      "Zaten çalışan bir sayaç var."
    );

    return;
  }

  const subject =
    subjects[index];

  timer = {

    id:index,

    subject:subject,

    total:total,

    startedAt:Date.now(),

    elapsed:0,

    remain:total,

    running:true

  };


  localStorage.setItem(
    "kpss_active_timer",
    JSON.stringify(timer)
  );


  document.getElementById(
    "timerSubject"
  ).textContent =
    subject;


  document.getElementById(
    "timerOverlay"
  ).style.display =
    "flex";


  updateTimer();


  timerInterval =
    setInterval(
      updateTimer,
      250
    );


  startNotificationTicker();


  sendNotification(

    "KPSS Odak Merkezi 🎯",

    `${subject} başladı • ${formatTime(total)} süre`

  );

}


/* =========================
   TIMER HESAPLA
========================= */

function calculateTimer(){

  if(!timer || !timer.running){

    return;
  }

  const elapsed =
    Math.min(
      timer.total,
      Math.max(
        0,
        (Date.now() -
        timer.startedAt) / 1000
      )
    );

  timer.elapsed =
    elapsed;

  timer.remain =
    Math.max(
      0,
      timer.total -
      elapsed
    );
}


/* =========================
   TIMER GÜNCELLE
========================= */

function updateTimer(){

  if(!timer) return;

  calculateTimer();

  document.getElementById(
    "bigTimer"
  ).textContent =
    formatTime(
      Math.ceil(timer.remain)
    );


  localStorage.setItem(
    "kpss_active_timer",
    JSON.stringify(timer)
  );


  if(timer.remain <= 0){

    finishTimer();

  }

}


/* =========================
   BİLDİRİM GÜNCELLEYİCİ
========================= */

function startNotificationTicker(){

  stopNotificationTicker();


  notificationInterval =
    setInterval(()=>{

      if(
        !timer ||
        !timer.running
      ){

        return;
      }

      calculateTimer();


      if(timer.remain <= 0){

        return;
      }


      /*
       * Aynı bildirimi sürekli spamlamamak
       * için 1 dakikada bir güncelliyoruz.
       *
       * Sayfa arka planda tamamen askıya alınırsa
       * tarayıcı bu intervali durdurabilir.
       */

      sendNotification(

        "KPSS Odak Merkezi 🎯",

        `${timer.subject} • ${formatTime(Math.ceil(timer.remain))} kaldı`

      );

    },60000);

}


function stopNotificationTicker(){

  if(notificationInterval){

    clearInterval(
      notificationInterval
    );

    notificationInterval = null;
  }

}


/* =========================
   TIMER BİTİR
========================= */

function finishTimer(){

  if(!timer) return;


  calculateTimer();


  const subject =
    timer.subject;

  const total =
    timer.total;


  const data =
    getDay();


  data.time[subject] =
    (data.time[subject] || 0) +
    total;


  saveDay(data);


  clearInterval(
    timerInterval
  );

  timerInterval = null;


  stopNotificationTicker();


  timer.running = false;


  localStorage.removeItem(
    "kpss_active_timer"
  );


  sendNotification(

    "🎯 Çalışma tamamlandı!",

    `${subject} çalışması tamamlandı. ${formatMinutes(total)} çalıştın.`

  );


  timer = null;


  document.getElementById(
    "timerOverlay"
  ).style.display =
    "none";


  updateAll();


  alert(
    `🎯 ${subject} çalışması tamamlandı!`
  );

}


/* =========================
   TIMER DURDUR
========================= */

function stopTimer(){

  if(!timer) return;


  calculateTimer();


  const elapsed =
    Math.floor(
      timer.elapsed
    );


  if(elapsed > 0){

    const data =
      getDay();


    data.time[timer.subject] =
      (data.time[timer.subject] || 0) +
      elapsed;


    saveDay(data);

  }


  clearInterval(
    timerInterval
  );

  timerInterval = null;


  stopNotificationTicker();


  localStorage.removeItem(
    "kpss_active_timer"
  );


  timer = null;


  document.getElementById(
    "timerOverlay"
  ).style.display =
    "none";


  updateAll();

}


/* =========================
   TIMER GERİ YÜKLE
========================= */

function restoreTimer(){

  const raw =
    localStorage.getItem(
      "kpss_active_timer"
    );


  if(!raw) return;


  try{

    const saved =
      JSON.parse(raw);


    if(
      !saved ||
      !saved.running
    ){

      return;
    }


    timer = saved;


    calculateTimer();


    if(timer.remain <= 0){

      finishTimer();

      return;
    }


    document.getElementById(
      "timerSubject"
    ).textContent =
      timer.subject;


    document.getElementById(
      "timerOverlay"
    ).style.display =
      "flex";


    updateTimer();


    timerInterval =
      setInterval(
        updateTimer,
        250
      );


    startNotificationTicker();


  }catch(error){

    localStorage.removeItem(
      "kpss_active_timer"
    );

  }

}

async function openFloatingTimer(){

      win.document.createElement(

      );


    subject.id = "subject";


    const time =
      win.document.createElement(
        "div"
      );


    time.id = "time";


    win.document.body.appendChild(
      subject
    );

      if(
        !timer ||
        !timer.running
      ){

      subject.textContent =
        timer.subject;


      time.textContent =

        );

    }

      setInterval(
        updatePiP,
        250
      );


    win.addEventListener(
      "pagehide",
      ()=>{
        clearInterval(
          pipInterval
        );
      }
    );


    updatePiP();


  }catch(error){

    console.log(
      "Yüzen sayaç açılamadı:",
      error
    );

    alert(
      "Yüzen sayaç açılamadı. Tarayıcın bu özelliği desteklemiyor olabilir."
    );

  }

}

function updateToday(){

  const data =
    getDay();


  let totalTime = 0;

  let totalQuestions = 0;


  subjects.forEach(
    subject=>{

      totalTime +=
        Number(
          data.time[subject] || 0
        );


      totalQuestions +=
        Number(
          data.questions[subject] || 0
        );

    }
  );


  document.getElementById(
    "todayTime"
  ).textContent =
    formatMinutes(totalTime);


  document.getElementById(
    "todayQuestions"
  ).textContent =
    totalQuestions;


  const percent =
    Math.min(
      100,
      (totalTime /
      DAILY_GOAL) *
      100
    );


  document.getElementById(
    "progressBar"
  ).style.width =
    percent + "%";


  document.getElementById(
    "progressText"
  ).textContent =
    "%" +
    Math.round(percent) +
    " tamamlandı";

}


/* =========================
   TAKVİM
========================= */

function renderCalendar(){

  const container =
    document.getElementById(
      "calendar"
    );


  container.innerHTML = "";


  const today =
    new Date();


  for(let i=0;i<28;i++){

    const d =
      new Date(today);


    d.setDate(
      today.getDate() + i
    );


    const key =
      d.getFullYear() +
      "-" +
      String(
        d.getMonth()+1
      ).padStart(2,"0") +
      "-" +
      String(
        d.getDate()
      ).padStart(2,"0");


    const data =
      getDay(key);


    let total = 0;


    subjects.forEach(
      s=>{
        total +=
          Number(
            data.time[s] || 0
          );
      }
    );


    const div =
      document.createElement(
        "div"
      );


    div.className = "day";


    if(key === todayKey()){

      div.classList.add(
        "today"
      );

    }


    if(total >= DAILY_GOAL){

      div.classList.add(
        "goal"
      );

    }else if(total > 0){

      div.classList.add(
        "partial"
      );

    }


    div.innerHTML = `

      <strong>
        ${d.getDate()}
      </strong>

      <br>

      ${formatMinutes(total)}

    `;


    container.appendChild(
      div
    );

  }

}

========================= */

function updateStats(){

  let totalTime = 0;

  let totalQuestions = 0;

  let studyDays = 0;

  let longestDay = 0;


  for(let i=0;i<28;i++){

    const d =
      new Date();


    d.setDate(
      d.getDate() - i
    );


    const key =
      d.getFullYear() +
      "-" +
      String(
        d.getMonth()+1
      ).padStart(2,"0") +
      "-" +
      String(
        d.getDate()
      ).padStart(2,"0");


    const data =
      getDay(key);


    let dayTime = 0;


    subjects.forEach(
      s=>{

        dayTime +=
          Number(
            data.time[s] || 0
          );


        totalQuestions +=
          Number(
            data.questions[s] || 0
          );

      }
    );


    totalTime +=
      dayTime;


    if(dayTime > 0){

      studyDays++;

    }


    longestDay =
      Math.max(
        longestDay,
        dayTime
      );

  }


  document.getElementById(
    "totalTime"
  ).textContent =
    formatMinutes(totalTime);


  document.getElementById(
    "totalQuestions"
  ).textContent =
    totalQuestions;


  document.getElementById(
    "studyDays"
  ).textContent =
    studyDays;


  document.getElementById(
    "longestDay"
  ).textContent =
    formatMinutes(longestDay);

}

function updateAll(){

  updateToday();

  renderCalendar();

  updateStats();

}
