<script>

/* =====================================================
   JSON DOSYASINDAN SORULARI ÇEK
   ===================================================== */

let sorular = [];

let mevcutSoru = 0;

let cevaplar = {};

let testBasladi = false;

let saniye = 0;

let timerInterval;


/* =====================================================
   JSON YÜKLE
   ===================================================== */

async function sorulariYukle() {

    try {

        const cevap = await fetch("sorular.json");

        if (!cevap.ok) {

            throw new Error(
                "sorular.json bulunamadı."
            );

        }

        sorular = await cevap.json();


        document.getElementById(
            "totalQuestionText"
        ).textContent = sorular.length;


    } catch (hata) {

        alert(
            "Sorular yüklenemedi!\n\n" +
            "index.html ile sorular.json aynı klasörde olmalı."
        );

        console.error(hata);
    }
}


/* =====================================================
   TESTİ BAŞLAT
   ===================================================== */

function startTest() {

    if (sorular.length === 0) {

        alert("Henüz soru yüklenmedi.");

        return;
    }


    mevcutSoru = 0;

    cevaplar = {};

    saniye = 0;

    testBasladi = true;


    document.getElementById(
        "startScreen"
    ).style.display = "none";


    document.getElementById(
        "resultScreen"
    ).style.display = "none";


    document.getElementById(
        "testScreen"
    ).style.display = "block";


    clearInterval(timerInterval);


    timerInterval = setInterval(() => {

        saniye++;

        let dakika =
            Math.floor(saniye / 60);

        let saniyeKismi =
            saniye % 60;


        document.getElementById(
            "timer"
        ).textContent =
            "⏱️ " +
            String(dakika).padStart(2,"0") +
            ":" +
            String(saniyeKismi).padStart(2,"0");

    }, 1000);


    soruyuGoster();
}


/* =====================================================
   SORUYU GÖSTER
   ===================================================== */

function soruyuGoster() {

    const soru =
        sorular[mevcutSoru];


    const toplam =
        sorular.length;


    document.getElementById(
        "questionCounter"
    ).textContent =
        `${mevcutSoru + 1} / ${toplam}`;


    const yuzde =
        Math.round(
            ((mevcutSoru + 1) / toplam) * 100
        );


    document.getElementById(
        "percent"
    ).textContent =
        yuzde + "%";


    document.getElementById(
        "progressBar"
    ).style.width =
        yuzde + "%";


    document.getElementById(
        "questionNumber"
    ).textContent =
        "Soru " + (mevcutSoru + 1);


    document.getElementById(
        "questionText"
    ).textContent =
        soru.soru;


    const options =
        document.getElementById("options");


    options.innerHTML = "";


    const harfler =
        ["A", "B", "C", "D", "E"];


    harfler.forEach(harf => {

        if (
            soru.secenekler[harf] === undefined
        ) {
            return;
        }


        const div =
            document.createElement("div");


        div.className = "option";


        div.innerHTML = `

            <div class="option-letter">
                ${harf}
            </div>

            <div class="option-text">
                ${soru.secenekler[harf]}
            </div>

        `;


        div.onclick = () =>
            cevapSec(harf, div);


        options.appendChild(div);

    });


    document.getElementById(
        "explanation"
    ).style.display = "none";


    document.getElementById(
        "explanation"
    ).textContent = "";


    document.getElementById(
        "nextButton"
    ).disabled = true;


    document.getElementById(
        "backButton"
    ).disabled =
        mevcutSoru === 0;


    /*
       Daha önce cevaplandıysa
       cevabı tekrar göster
    */

    if (
        cevaplar[mevcutSoru]
    ) {

        const eskiCevap =
            cevaplar[mevcutSoru];


        const secenekler =
            document.querySelectorAll(
                ".option"
            );


        secenekler.forEach(option => {

            const harf =
                option
                .querySelector(
                    ".option-letter"
                )
                .textContent
                .trim();


            option.classList.add(
                "disabled"
            );


            if (
                harf === soru.cevap
            ) {

                option.classList.add(
                    "correct"
                );

            }


            if (
                harf === eskiCevap &&
                eskiCevap !== soru.cevap
            ) {

                option.classList.add(
                    "wrong"
                );

            }

        });


        document.getElementById(
            "explanation"
        ).textContent =
            "💡 " + soru.aciklama;


        document.getElementById(
            "explanation"
        ).style.display = "block";


        document.getElementById(
            "nextButton"
        ).disabled = false;
    }

}


/* =====================================================
   CEVAP SEÇ
   ===================================================== */

function cevapSec(harf, secilenElement) {

    if (
        cevaplar[mevcutSoru]
    ) {
        return;
    }


    const soru =
        sorular[mevcutSoru];


    cevaplar[mevcutSoru] =
        harf;


    const secenekler =
        document.querySelectorAll(
            ".option"
        );


    secenekler.forEach(option => {

        option.classList.add(
            "disabled"
        );


        const optionHarf =
            option
            .querySelector(
                ".option-letter"
            )
            .textContent
            .trim();


        if (
            optionHarf === soru.cevap
        ) {

            option.classList.add(
                "correct"
            );

        }


        if (
            optionHarf === harf &&
            harf !== soru.cevap
        ) {

            option.classList.add(
                "wrong"
            );

        }

    });


    document.getElementById(
        "explanation"
    ).textContent =
        "💡 " + soru.aciklama;


    document.getElementById(
        "explanation"
    ).style.display = "block";


    document.getElementById(
        "nextButton"
    ).disabled = false;

}


/* =====================================================
   SONRAKİ SORU
   ===================================================== */

function nextQuestion() {

    if (
        mevcutSoru <
        sorular.length - 1
    ) {

        mevcutSoru++;

        soruyuGoster();

    } else {

        testiBitir();

    }

}


/* =====================================================
   ÖNCEKİ SORU
   ===================================================== */

function previousQuestion() {

    if (mevcutSoru > 0) {

        mevcutSoru--;

        soruyuGoster();

    }

}


/* =====================================================
   TESTİ BİTİR
   ===================================================== */

function testiBitir() {

    clearInterval(timerInterval);


    let dogru = 0;

    let yanlis = 0;

    let bos = 0;


    sorular.forEach(
        (soru, index) => {

            const cevap =
                cevaplar[index];


            if (!cevap) {

                bos++;

            }

            else if (
                cevap === soru.cevap
            ) {

                dogru++;

            }

            else {

                yanlis++;

            }

        }
    );


    /*
       KPSS sistemi:
       4 yanlış = 1 doğru
    */

    const net =
        dogru -
        (yanlis / 4);


    document.getElementById(
        "correctResult"
    ).textContent =
        dogru;


    document.getElementById(
        "wrongResult"
    ).textContent =
        yanlis;


    document.getElementById(
        "emptyResult"
    ).textContent =
        bos;


    document.getElementById(
        "netResult"
    ).textContent =
        net.toFixed(2);


    document.getElementById(
        "testScreen"
    ).style.display =
        "none";


    document.getElementById(
        "resultScreen"
    ).style.display =
        "block";

}


/* =====================================================
   YENİDEN BAŞLAT
   ===================================================== */

function restartTest() {

    clearInterval(timerInterval);

    mevcutSoru = 0;

    cevaplar = {};

    saniye = 0;


    document.getElementById(
        "resultScreen"
    ).style.display =
        "none";


    document.getElementById(
        "startScreen"
    ).style.display =
        "block";


    document.getElementById(
        "timer"
    ).textContent =
        "⏱️ 00:00";

}


/* =====================================================
   SAYFA AÇILINCA JSON'U YÜKLE
   ===================================================== */

sorulariYukle();

</script>
